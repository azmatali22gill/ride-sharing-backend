const io = require("socket.io-client");
const socket = io("http://localhost:3010/realtime");

const DRIVER_ID = "DRIVER_ID";  // <-- YAHAN APNA DRIVER_ID LAGAO
const RIDE_ID = "RIDE_ID";      // <-- YAHAN APNA RIDE_ID LAGAO

socket.on("connect", () => {
  console.log("✅ WebSocket Connected!");

  // Driver register karwao
  socket.emit("driver:register", { driverId: DRIVER_ID });
  console.log("📤 Sent: driver:register");

  // Ride room join karo
  socket.emit("ride:join", { rideId: RIDE_ID });
  console.log("📤 Sent: ride:join");
});

// Server se aane wale events suno:
socket.on("ride:offer", (data) => {
  console.log("📩 Received ride:offer →", JSON.stringify(data, null, 2));
});

socket.on("ride:accepted", (data) => {
  console.log("📩 Received ride:accepted →", data);
});

socket.on("ride:status", (data) => {
  console.log("📩 Received ride:status →", data);
});

socket.on("ride:completed", (data) => {
  console.log("📩 Received ride:completed →", data);
});

socket.on("ride:cancelled", (data) => {
  console.log("📩 Received ride:cancelled →", data);
});

socket.on("ride:expired", (data) => {
  console.log("📩 Received ride:expired →", data);
});

socket.on("disconnect", () => {
  console.log("❌ Disconnected");
});

// Driver location update bhejne ke liye (simulate GPS):
setTimeout(() => {
  socket.emit("driver:location", {
    driverId: DRIVER_ID,
    rideId: RIDE_ID,
    lat: 24.8610,
    lng: 67.0015
  });
  console.log("📤 Sent: driver:location update");
}, 2000);

#!/bin/bash

BASE="http://localhost:3010"

echo "=== 1. Passenger Register ==="
PASS=$(curl -s -X POST $BASE/passengers \
  -H "Content-Type: application/json" \
  -d '{"name":"Ali Raza","email":"ali@test.com","phone":"03001234567"}')
echo $PASS | grep -o '"_id":"[^"]*"'
PASS_ID=$(echo $PASS | grep -o '"_id":"[^"]*"' | cut -d'"' -f4)
echo "PASSENGER_ID: $PASS_ID"

echo ""
echo "=== 2. Driver Onboard ==="
DRV=$(curl -s -X POST $BASE/drivers \
  -H "Content-Type: application/json" \
  -d '{"name":"Ahmed Khan","email":"ahmed@test.com","phone":"03007654321","age":28,"gender":"Male","vehicle":{"make":"Toyota","model":"Corolla","color":"White","plateNumber":"LEA-123","capacity":4,"category":"ECONOMY"}}')
DRV_ID=$(echo $DRV | grep -o '"_id":"[^"]*"' | head -1 | cut -d'"' -f4)
echo "DRIVER_ID: $DRV_ID"

echo ""
echo "=== 3. Verify Driver ==="
curl -s -X PATCH $BASE/drivers/$DRV_ID/verification \
  -H "Content-Type: application/json" \
  -d '{"verificationStatus":"VERIFIED"}'
echo " (VERIFIED)"

echo ""
echo "=== 4. Driver Available ==="
curl -s -X PATCH $BASE/drivers/$DRV_ID/status \
  -H "Content-Type: application/json" \
  -d '{"status":"AVAILABLE"}'
echo " (AVAILABLE)"

echo ""
echo "=== 5. Update Location ==="
curl -s -X PATCH $BASE/drivers/$DRV_ID/location \
  -H "Content-Type: application/json" \
  -d '{"lat":24.8607,"lng":67.0011}'
echo " (LOCATION UPDATED)"

echo ""
echo "=== 6. Request Ride ==="
RIDE=$(curl -s -X POST $BASE/rides \
  -H "Content-Type: application/json" \
  -d "{\"passengerId\":\"$PASS_ID\",\"pickupLocation\":{\"lat\":24.8607,\"lng\":67.0011,\"address\":\"Rashid Minhas\"},\"destination\":{\"lat\":24.8918,\"lng\":67.0297,\"address\":\"DHA Phase 2\"},\"rideType\":\"ECONOMY\"}")
RIDE_ID=$(echo $RIDE | grep -o '"_id":"[^"]*"' | head -1 | cut -d'"' -f4)
echo "RIDE_ID: $RIDE_ID"
echo "Fare: $(echo $RIDE | grep -o '"total":[0-9]*')"

echo ""
echo "=== 7. Accept Ride ==="
curl -s -X PATCH $BASE/rides/$RIDE_ID/accept \
  -H "Content-Type: application/json" \
  -d "{\"driverId\":\"$DRV_ID\"}"
echo " (ACCEPTED)"

echo ""
echo "=== 8. Arriving ==="
curl -s -X PATCH $BASE/rides/$RIDE_ID/arriving \
  -H "Content-Type: application/json" \
  -d "{\"driverId\":\"$DRV_ID\"}"
echo " (ARRIVING)"

echo ""
echo "=== 9. Start Ride ==="
curl -s -X PATCH $BASE/rides/$RIDE_ID/start \
  -H "Content-Type: application/json" \
  -d "{\"driverId\":\"$DRV_ID\"}"
echo " (STARTED)"

echo ""
echo "=== 10. Complete Ride ==="
curl -s -X PATCH $BASE/rides/$RIDE_ID/complete \
  -H "Content-Type: application/json" \
  -d "{\"driverId\":\"$DRV_ID\"}"
echo " (COMPLETED)"

echo ""
echo "=== 11. Submit Rating ==="
curl -s -X POST $BASE/ratings \
  -H "Content-Type: application/json" \
  -d "{\"rideId\":\"$RIDE_ID\",\"raterType\":\"PASSENGER\",\"fromUserId\":\"$PASS_ID\",\"toUserId\":\"$DRV_ID\",\"rating\":5,\"review\":\"Great ride!\"}"
echo " (RATED)"

echo ""
echo "=== 12. History ==="
curl -s $BASE/rides/history/passenger/$PASS_ID | grep -o '"totalSpending":[0-9]*'
curl -s $BASE/rides/history/driver/$DRV_ID | grep -o '"totalEarnings":[0-9]*'

echo ""
echo "✅ DONE! Full lifecycle tested successfully."

    lat += 0.0001;

      driverId: DRIVER_ID,

      lat: lat,
      lng: 67.0011
    });

});
